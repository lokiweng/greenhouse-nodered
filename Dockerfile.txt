FROM nodered/node-red:latest

USER root

# Install the dashboard nodes
WORKDIR /usr/src/node-red
RUN npm install node-red-dashboard

# The [ "src", "dest" ] format is specifically for handling spaces
COPY ["current version flows.json", "/data/current version flows.json"]

# Ensure permissions are correct for the node-red user
RUN chown -R node-red:node-red /data

USER node-red

# Use quotes for the environment variable to handle the spaces
ENV FLOWS="current version flows.json"

# Redirect the root page to the dashboard UI automatically
ENV NODE_OPTIONS="--max-old-space-size=512"


RUN sed -i 's/\/\/httpRoot: \x27\/\x27,/httpRoot: \x27\/\x27,/' /usr/src/node-red/node_modules/node-red/settings.js \
    && sed -i 's/ui: { path: "ui" }/ui: { path: "" }/' /usr/src/node-red/node_modules/node-red-dashboard/index.js || true

EXPOSE 1880
