FROM nodered/node-red:latest

USER root

# Install the dashboard nodes
WORKDIR /usr/src/node-red
RUN npm install node-red-dashboard

# Copy the file AND rename it to flows.json inside the container
# This permanently fixes the "spaces in filename" error
COPY ["current version flows.json", "/data/flows.json"]

RUN chown -R node-red:node-red /data

USER node-red

# Tell Node-RED to load the renamed file
ENV FLOWS=flows.json

# Tell Railway exactly which port to use
ENV PORT=1880
EXPOSE 1880
